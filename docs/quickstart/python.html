5. Functions

python
def describe(name, level=3):        # level has a default
    label = "expert" if level >= 5 else "working"
    return f"{name}: {label} (L{level})"

describe("Priya Raman")             # "Priya Raman: working (L3)"
describe("Priya Raman", level=5)    # "Priya Raman: expert (L5)"

return hands a value back; print only shows it. Full detail: Functions.

7. Classes (OOP) in 20 seconds

python
class Employee:
    def __init__(self, employee_id, name, skills):  # setup, runs on creation
        self.employee_id = employee_id              # this object's data
        self.name = name
        self.skills = skills

    def level(self, skill):
        """0, not an error, when they have never used it."""
        return self.skills.get(skill, 0)

priya = Employee("E1002", "Priya Raman", {"Python": 4, "SQL": 5})
priya.level("Python")     # 4
priya.level("Kafka")      # 0

self = "this object." Principles: encapsulation, inheritance, polymorphism, abstraction. Full detail: Objects & Classes.

A @dataclass writes __init__ for you — that is what the Skills Matcher actually uses:

python
from dataclasses import dataclass, field

@dataclass
class Employee:
    employee_id: str
    name: str
    location: str
    skills: dict[str, int] = field(default_factory=dict)   # never `= {}` — see gotchas

    def level(self, skill: str) -> int:
        return self.skills.get(skill, 0)

8. Generators & decorators

python
import functools

# generator — yields values one at a time (memory-friendly)
def candidates_for(matches):
    """Only the people who are not blocked, best score first."""
    for match in sorted(matches, key=lambda m: m.score, reverse=True):
        if not match.blockers:
            yield match

# decorator — wraps a function to add behaviour (wraps preserves its name)
def log_calls(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        result = func(*args, **kwargs)
        print(f"  · {func.__name__} -> {result!r}")
        return result
    return wrapper

@log_calls
def add(employee_id, name, location, skills):
    ...

Lazy is the point. next(bench.candidates_for(j2001)) scores only as many people as it takes to find the first viable one — it does not build a list of everybody first.

You'll see @ everywhere later (@app.get, @mcp.tool, @tool). These are the exact candidates_for() generator and log_calls decorator used in python/skills_matcher.py. Full detail: Generators · Decorators.

9. Error handling

python
class BenchError(Exception):
    """Your own error type — clearer to callers than a bare ValueError."""

def add(employee_id, level):
    if not 1 <= level <= 5:
        raise BenchError(f"level must be 1..5, got {level}")

try:
    add("E1002", level=9)
except BenchError as exc:
    print(f"  ✗ rejected: {exc}")     # ✗ rejected: level must be 1..5, got 9
finally:
    print("always runs")

Catch the specific error you expect (ValueError, KeyError, or one you defined). A custom exception lets a caller handle your failures without accidentally swallowing unrelated bugs. Full detail: Handling Errors.
